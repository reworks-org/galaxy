87 {
88 if (!cursor.empty())

94 auto data = fs.read_binary(cursor);
95 if (!data.empty())

97 auto surface = STBIMG_LoadFromMemory(data.data(), static_cast<int>(data.size()));
98 if (surface)
99 {
100 if (m_cursor_bitmap)

102 sl::destroy_bitmap(m_cursor_bitmap);
103 }
104
105 m_cursor_bitmap = sl::load_bitmap_from_surface(surface);

111 }
112
