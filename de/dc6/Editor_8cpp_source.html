410 {
411 std::ifstream ifs {path, std::ifstream::in | std::ifstream::binary | std::ifstream::ate};
412
413 if (ifs.good())
414 {
415 meta::vector<char> buffer;
416

419
420 ifs.seekg(0, std::ifstream::beg);
421 ifs.read(&buffer[0], size);
422 ifs.close();
423
424 m_current_project_path = std::filesystem::path(path);

426 auto data = std::string(buffer.begin(), buffer.end());
